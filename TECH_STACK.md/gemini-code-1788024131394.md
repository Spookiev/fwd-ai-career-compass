# Technical Stack & System Architecture Specification

## 1. High-Level Technology Matrix

| Layer | Technology | Version | Purpose / Justification |
| :--- | :--- | :--- | :--- |
| **Frontend Framework** | React 18 / TypeScript | React ^18.3, TS ^5.5 | Type-safe enterprise dashboard with high performance. |
| **Build Tool** | Vite | ^5.4 | Sub-second HMR and optimized production bundling. |
| **Styling & UI** | Tailwind CSS | ^3.4 | Custom dark mode glassmorphism with high visual density. |
| **Icons** | Lucide React | ^0.344 | Consistent, lightweight vector iconography. |
| **Data Visualizations**| Recharts | ^2.12 | Responsive Bar, Area, Pie, and Radar charting. |
| **Document Processing**| pdfjs-dist | ^4.0 | Client-side parsing of uploaded PDF resumes. |
| **Effects & Feedback** | Canvas Confetti | ^1.9 | Positive reinforcement on milestone completion. |
| **Backend API** | Node.js / Express | Node ^18.x, Express ^4.19 | API routing and middleware proxy for AI services. |
| **Authentication** | Firebase Authentication | ^10.x | Google OAuth 2.0 and Email/Password with verification. |
| **Database** | Cloud Firestore | ^10.x | NoSQL document storage with real-time sync. |
| **AI / NLP Engine** | Google Gen AI SDK (`@google/genai`) | Gemini 2.5 / 3.0 Pro | LLM career matching, ATS auditing, and interview evaluation. |

## 2. Cloud Firestore Database Schema (FWD 2.0)

```typescript
// /users/{userId}
interface UserDocument {
  uid: string;
  email: string;
  displayName: string;
  role: 'student' | 'faculty';
  avatar: {
    id: string;
    category: 'animals' | 'robots' | 'fantasy' | 'anime' | 'professional' | 'minimalist';
    unlockedTier: 'foundation' | 'associate' | 'advanced';
  };
  createdAt: string;
}

// /evidence/{userId}
interface EvidenceDocument {
  userId: string;
  // Professional
  resume?: {
    uploadedAt: string;
    rawText: string;
    extractedSkills: string[];
    extractedExperienceYears: number;
    extractedProjects: Array<{ title: string; tech: string[]; description: string }>;
  };
  linkedInUrl?: string;
  portfolioUrl?: string;
  // Technical Evidence
  github?: {
    username: string;
    connected: boolean;
    repoCount: number;
    topLanguages: Record<string, number>; // e.g. { TypeScript: 60, Python: 40 }
    totalStars: number;
    activityScore: number;
  };
  leetcode?: {
    username: string;
    connected: boolean;
    problemsSolved: { easy: number; medium: number; hard: number };
    contestRating?: number;
  };
  hackerrank?: { username: string; badges: string[] };
  // Academic Evidence
  academics?: {
    university: string;
    degree: string;
    semester: string;
    cgpa: number;
    strongSubjects: string[];
    weakSubjects: string[];
  };
  // Self-Narrative
  selfIntroduction?: {
    rawText: string;
    extractedInterests: string[];
    inferredStrengths: string[];
  };
  // Certifications
  certificates: Array<{
    id: string;
    name: string;
    provider: 'NPTEL' | 'Coursera' | 'Udemy' | 'Google' | 'Other';
    issueDate: string;
    credentialUrl?: string;
  }>;
}

// /skillProfiles/{userId}
interface TriangulatedSkillProfile {
  userId: string;
  skills: Record<string, {
    name: string;
    claimedLevel?: 'Beginner' | 'Intermediate' | 'Advanced';
    githubEvidenceScore?: number; // 0-100
    assessmentScore?: number;     // 0-100
    projectEvidenceScore?: number; // 0-100
    compositeConfidence: 'Low' | 'Medium' | 'High';
    effectiveTier: 'Foundation' | 'Associate' | 'Advanced';
    growthHistory: Array<{ date: string; score: number }>;
  }>;
  interestVector: Record<string, number>; // e.g. { design: 0.85, analytics: 0.60, coding: 0.40 }
  workStyle: {
    collaboration: number; // 0 (Solo) to 100 (Team)
    structure: number;     // 0 (Flexible) to 100 (Structured)
    orientation: number;   // 0 (Deep Tech) to 100 (People Focused)
    execution: number;     // 0 (Build) to 100 (Analyze)
  };
}

// /careerMatches/{userId}
interface CareerPossibilityMap {
  userId: string;
  updatedAt: string;
  matches: Array<{
    roleId: string;
    title: string;
    family: 'Technology' | 'Product & Business' | 'Creative' | 'Communication';
    compatibilityScore: number; // 0-100
    matchType: 'Strong Match' | 'Emerging Match' | 'Exploratory Match';
    whyItSuitsYou: string[];
    whatMayChallengeYou: string[];
    skillGaps: Array<{ skill: string; criticality: 'Blocker' | 'Priority' | 'Differentiator' }>;
    miniTrial: {
      title: string;
      durationDays: 7;
      tasks: Array<{ day: number; task: string; testGoal: string }>;
    };
  }>;
}

// /careerPresence/{userId}
interface CareerPresenceDocument {
  userId: string;
  overallScore: number; // 0-100
  breakdown: {
    github: { score: number; checklist: { bio: boolean; readme: boolean; pinnedRepos: boolean; liveLinks: boolean } };
    linkedIn: { score: number; checklist: { headline: boolean; about: boolean; experience: boolean; skills: boolean } };
    resumeAts: { score: number; missingKeywords: string[]; quantifiedBullets: boolean };
    portfolio: { score: number; responsive: boolean; caseStudies: boolean };
    certifications: { score: number; count: number };
  };
}

// /wellbeing/{userId}
interface WellbeingDocument {
  userId: string;
  currentStreak: number;
  streakFrozen: boolean;
  dailyLogs: Array<{
    date: string;
    moodEmoji: string; // e.g. "😊" | "🙂" | "😐" | "😕" | "😫"
    vibeCard: string;  // e.g. "🌱" | "🔥" | "🌙" | "🌧️"
    preEnergyLevel: number;  // 1 to 5
    postEnergyLevel?: number; // 1 to 5
    studyDurationMinutes: number;
  }>;
  fatigueScore: number; // 0-100
  pacingRecommendation: 'Normal' | 'Slow Down & Break' | 'Recovery Day';
}
```
